cmake_minimum_required(VERSION 3.24)
project(install_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(TEST_COMPONENTS)
if(TEST_SCHEMA_JSON)
	list(APPEND TEST_COMPONENTS schema_json)
endif()
if(TEST_HEADER_JSON)
	list(APPEND TEST_COMPONENTS header_json)
endif()
find_package(vmf REQUIRED CONFIG OPTIONAL_COMPONENTS schema_json header_json)
foreach(component schema_json header_json)
	string(TOUPPER "TEST_${component}" option)
	if(${option})
		if(NOT vmf_${component}_FOUND OR NOT TARGET vmf::${component})
			message(FATAL_ERROR "Installed ${component} is unavailable")
		endif()
	elseif(vmf_${component}_FOUND OR TARGET vmf::${component})
		message(FATAL_ERROR "Unbuilt ${component} was advertised")
	endif()
endforeach()
find_package(vmf REQUIRED CONFIG COMPONENTS ${TEST_COMPONENTS})
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE vmf::vmf)
foreach(component IN LISTS TEST_COMPONENTS)
	target_link_libraries(consumer PRIVATE vmf::${component})
	string(TOUPPER "TEST_${component}" definition)
	target_compile_definitions(consumer PRIVATE "${definition}")
endforeach()
